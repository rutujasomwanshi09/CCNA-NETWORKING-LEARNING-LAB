LAB 35 - FLOATING STATIC ROUTING CONFIGURATION USING CISCO PACKET TRACER 
=================================

## CONFIGURE ROUTER - KE-R1


enable
configure terminal

interface g0/0
ip address 192.168.10.1 255.255.255.0
no shutdown
exit

interface g0/1
ip address 10.10.10.1 255.255.255.252
no shutdown
exit

interface g0/2
ip address 40.40.40.1 255.255.255.252
no shutdown
exit

ip route 192.168.20.0 255.255.255.0 g0/1
ip route 192.168.20.0 255.255.255.0 40.40.40.2 130

end
write memory
